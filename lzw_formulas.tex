\section*{LZW Compression Formula Sheet}

\[
\text{Bit Width} = \lceil \log_2(9) \rceil = 4
\]

\[
\text{Original Size} = 12 \times 8 = 96\ \text{bits}
\]

\[
\text{Compressed Size} = 6 \times 4
= 24\ \text{bits}
\]

\[
\text{Savings} =
\left(1 - \frac{24}{96}\right) \times 100
= 75.0\%
\]
